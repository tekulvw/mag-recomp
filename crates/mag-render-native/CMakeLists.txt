cmake_minimum_required(VERSION 3.24)
project(mag_render_native LANGUAGES C CXX)
set(MAG_RENDER_NATIVE_BACKEND "none" CACHE STRING "Frontend-owned native backend")
add_library(mag_render_native_impl STATIC native/mag_render_native.c native/mag_render_native_api.cpp)
target_include_directories(mag_render_native_impl PUBLIC native)
target_compile_features(mag_render_native_impl PRIVATE c_std_11 cxx_std_17)
if(MSVC)
    target_compile_options(mag_render_native_impl PRIVATE /W4 /WX)
else()
    target_compile_options(mag_render_native_impl PRIVATE -Wall -Wextra -Werror)
endif()
if(MAG_RENDER_NATIVE_BACKEND STREQUAL "metal")
    if(NOT APPLE)
        message(FATAL_ERROR "Metal requires macOS")
    endif()
    enable_language(OBJC)
    target_sources(mag_render_native_impl PRIVATE native/mag_render_metal.m)
    target_compile_definitions(mag_render_native_impl PRIVATE MAG_RENDER_NATIVE_METAL)
    target_compile_options(mag_render_native_impl PRIVATE "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc>")
elseif(MAG_RENDER_NATIVE_BACKEND STREQUAL "vulkan")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(VULKAN_HEADERS REQUIRED vulkan)
    target_sources(mag_render_native_impl PRIVATE native/mag_render_vulkan.c)
    target_compile_definitions(mag_render_native_impl PRIVATE MAG_RENDER_NATIVE_VULKAN)
    target_include_directories(mag_render_native_impl PRIVATE ${VULKAN_HEADERS_INCLUDE_DIRS})
    target_compile_options(mag_render_native_impl PRIVATE ${VULKAN_HEADERS_CFLAGS_OTHER})
elseif(MAG_RENDER_NATIVE_BACKEND STREQUAL "d3d12")
    message(FATAL_ERROR "D3D12 adapter is pending its owned implementation")
elseif(NOT MAG_RENDER_NATIVE_BACKEND STREQUAL "none")
    message(FATAL_ERROR "Unknown native backend")
endif()
